  147 |     duration_ms: float,
  148 | ) -> float:
  149 |     """
  150 |     Soft HUMAN-VOCAL plausibility penalty.
  151 | 
  152 |     Important musical rule:
  153 | 
  154 |         THIRDS AND BELOW (<= 4 semitones)
  155 |         receive essentially NO penalty as ordinary rapid
  156 |         melodic movement.
  157 | 
  158 |     Above a third, increasing interval size requires increasing time.
  159 | 
  160 |     This is NOT a validity decision.
  161 |     It is a diagnostic score.
  162 | 
  163 |     Rough interpretation:
  164 | 
  165 |         0.0   ordinary / unsurprising
  166 |         0.0-1 mildly demanding
  167 |         1-2   suspicious
  168 |         2-4   strongly suspicious
  169 |         >4    extremely implausible as an instantaneous target change
  170 | 
  171 |     Continuous glides are treated separately through trajectory
  172 |     coherence, so a large total excursion is not automatically bad.
  173 |     """
  174 | 
  175 |     interval = abs(
  176 |         float(interval_semitones)
  177 |     )
  178 | 
  179 |     duration_ms = max(
  180 |         float(duration_ms),
  181 |         1.0,
  182 |     )
  183 | 
  184 |     # -------------------------------------------------------------
  185 |     # Thirds and below:
  186 |     # essentially no penalty, even rapidly.
  187 |     # -------------------------------------------------------------
  188 | 
  189 |     if interval <= 4.0:
  190 |         return 0.0
  191 | 
  192 |     # -------------------------------------------------------------
  193 |     # Above a third:
  194 |     #
  195 |     # We construct a smoothly increasing "minimum comfortable time".
  196 |     #
  197 |     #  5 st  ~ fourth
  198 |     #  7 st  ~ fifth
  199 |     #  9 st  ~ sixth
  200 |     # 12 st  ~ octave
  201 |     #
  202 |     # These are not hard human limits.
  203 |     # They are priors for suspicious ECKF jumps.
  204 |     # -------------------------------------------------------------
  205 |

  209 |         18.0

  217 |         required_ms
